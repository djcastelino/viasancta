! =============================================================================
! NetFlow for SILICON ONE (C9500X etc.) only
! No collect next-hop, no collect interface output/input (not supported).
! Use on CORE-2 / any C9500X. Same apply-to-interface pattern; change interfaces per device.
! =============================================================================

! --- Flow records (Silicon One compatible) ---
flow record MS-interface-Input-S1
 description Silicon One compatible; ingress keyed by input interface
 match ipv4 protocol
 match ipv4 source address
 match ipv4 destination address
 match transport source-port
 match transport destination-port
 match interface input
 collect counter bytes long
 collect counter packets long
!

flow record MS-interface-Output-S1
 description Silicon One compatible; egress keyed by output interface
 match ipv4 protocol
 match ipv4 source address
 match ipv4 destination address
 match transport source-port
 match transport destination-port
 match interface output
 collect counter bytes long
 collect counter packets long
!

! --- Flow exporter ---
flow exporter FlowExporter1
 destination 10.81.124.97
 source Loopback0
 transport udp 9998
!

! --- Flow monitors ---
flow monitor FlowMonitor-MS-INPUT
 record MS-interface-Input-S1
 exporter FlowExporter1
!

flow monitor FlowMonitor-MS-OUTPUT
 record MS-interface-Output-S1
 exporter FlowExporter1
!
